# --- Auction Core Mechanics ---
auctions:
  behavior:
    # Duración por defecto de una subasta en horas, si el jugador no especifica una al crearla.
    default_duration_hours: 24
    # Permitir que los jugadores establezcan un precio de "Comprar Ahora" para sus subastas.
    allow_buy_now: true
    # Permitir la creación y el funcionamiento de "Subastas Misteriosas" (lotes de ítems ocultos).
    allow_mystery_auctions: true
    # Cantidad mínima que una nueva puja debe superar sobre la puja actual.
    # Ayuda a evitar incrementos de puja demasiado pequeños.
    min_bid_increment: 10.0
    # Intervalo (en segundos) con el que el plugin revisará y procesará las subastas expiradas.
    expired_check_interval_seconds: 60 # Cada minuto

  limits_and_restrictions:
    # Número máximo de subastas que un jugador puede tener activas al mismo tiempo.
    # (Futura mejora: Límites basados en permisos).
    max_active_per_player: 5
    # Lista de materiales (ítems) que NO pueden ser subastados.
    # Utiliza los nombres oficiales de Material de Bukkit.
    item_blacklist:
      - "BEDROCK"
      - "COMMAND_BLOCK"
      - "STRUCTURE_VOID"
      - "BARRIER"
      - "KNOWLEDGE_BOOK"
      - "DEBUG_STICK"
      - "AIR" # Aunque el código ya previene esto, es bueno tenerlo explícito.

  fees_and_commissions:
    # Tarifa fija que se cobra al jugador por crear cualquier tipo de subasta.
    # Establecer en 0.0 para no cobrar ninguna tarifa de creación.
    creation_fee: 0.0 # Ejemplo: 50.0 para cobrar 50 de la moneda del servidor.

    commission:
      # Define cómo se calcula la comisión del servidor sobre las ventas exitosas.
      # Tipos disponibles:
      #   "flat_percentage": Un porcentaje fijo sobre el precio de venta final.
      #   "per_category_percentage": Diferentes porcentajes basados en el material del ítem vendido.
      #   "tiered_percentage": Porcentajes que varían según rangos del precio de venta.
      type: "flat_percentage"

      # Porcentaje de comisión por defecto (0-100).
      # Utilizado para "flat_percentage" y como fallback si una categoría o tramo no está definido.
      default_percentage: 5.0 # Representa una comisión del 5%

      # Configuración para el tipo "per_category_percentage".
      # Asocia nombres de Material de Bukkit (en mayúsculas) a un porcentaje de comisión.
      categories:
        DIAMOND_SWORD: 10.0
        ELYTRA: 15.0
        NETHERITE_INGOT: 7.5
        ENCHANTED_GOLDEN_APPLE: 12.0
        # Comisión para todos los demás ítems no listados específicamente.
        DEFAULT_OTHER: 4.0 # Fallback para ítems sin categoría específica

      # Configuración para el tipo "tiered_percentage".
      # Lista de tramos de precios. La comisión se basa en el primer tramo cuyo 'max_price' sea
      # igual o superior al precio de venta. El orden de los tramos es importante (de menor a mayor 'max_price').
      # Usa 'max_price: -1' para el último tramo, que cubrirá todos los precios superiores al tramo anterior.
      tiers:
        - { max_price: 100.0, percentage: 7.0 }    # Para ventas <= 100
        - { max_price: 1000.0, percentage: 5.0 }   # Para ventas > 100 y <= 1000
        - { max_price: 10000.0, percentage: 3.0 }  # Para ventas > 1000 y <= 10000
        - { max_price: -1, percentage: 1.5 }       # Para ventas > 10000
